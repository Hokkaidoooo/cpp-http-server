cmake_minimum_required(VERSION 3.16)
project(cpp_http_server VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_SERVER "构建 Linux HTTP 服务器" ON)
set(SANITIZER "none" CACHE STRING "none、address 或 thread")
set_property(CACHE SANITIZER PROPERTY STRINGS none address thread)
find_package(Threads REQUIRED)
add_library(project_options INTERFACE)
target_include_directories(project_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(project_options INTERFACE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(project_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SANITIZER STREQUAL "address")
    target_compile_options(project_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(project_options INTERFACE -fsanitize=address,undefined)
elseif(SANITIZER STREQUAL "thread")
    target_compile_options(project_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(project_options INTERFACE -fsanitize=thread)
elseif(NOT SANITIZER STREQUAL "none")
    message(FATAL_ERROR "SANITIZER 只能为 none、address 或 thread")
endif()
enable_testing()
foreach(test_name parser thread_pool)
    add_executable(test_${test_name} tests/test_${test_name}.cpp)
    target_link_libraries(test_${test_name} PRIVATE project_options)
    add_test(NAME ${test_name} COMMAND test_${test_name})
endforeach()
if(BUILD_SERVER)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "服务器需要 Linux epoll/eventfd/renameat2；仅协议测试可用 -DBUILD_SERVER=OFF")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(MONGOCXX REQUIRED IMPORTED_TARGET libmongocxx=3.9.0)
    pkg_check_modules(BSONCXX REQUIRED IMPORTED_TARGET libbsoncxx=3.9.0)
    pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium>=1.0.18)
    find_package(PNG REQUIRED)
    add_executable(myserver main.cpp)
    target_link_libraries(myserver PRIVATE project_options PkgConfig::MONGOCXX PkgConfig::BSONCXX PkgConfig::SODIUM PNG::PNG)
endif()
